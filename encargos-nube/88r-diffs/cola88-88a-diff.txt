+# Spotify Connect por perfil (88a)
+
+Esta carpeta deja dos receptores `go-librespot v0.10.2`: Jaime en `127.0.0.1:3678` y Pilar en
+`127.0.0.1:3679`. No expone la API en la red ni usa zeroconf/mDNS. El audio usa `pulseaudio`,
+que en la Pi es `pipewire-pulse`, y por tanto conserva el *sink* (salida) por defecto.
+
+## Instalación
+
+La cola instala el binario armado en `/usr/local/lib/go-librespot/v0.10.2/go-librespot` y deja
+`/usr/local/bin/go-librespot` como enlace simbólico a esa versión. Instala esta unidad en
+`~/.config/systemd/user/spotify@.service` y las dos plantillas en el volumen LUKS:
+
+| Origen | Destino | Dueño y modo |
+|---|---|---|
+| `config-jaime.yml` | `/opt/nodo/datos/spotify/jaime/config.yml` | `dcifu004:dcifu004 0600` (carpeta `0700`) |
+| `config-pilar.yml` | `/opt/nodo/datos/spotify/pilar/config.yml` | `dcifu004:dcifu004 0600` (carpeta `0700`) |
+| `spotify@.service` | `~/.config/systemd/user/spotify@.service` | `root:root 0644` |
+
+Después de que la cola haya desplegado todo, como `dcifu004` se habilitan y arrancan las dos
+instancias una sola vez:
+
+```bash
+systemctl --user daemon-reload && systemctl --user enable --now spotify@jaime spotify@pilar
+```
+
+## Vincular una cuenta (después de 88b)
+
+No vincules desde esta etapa. Mientras un perfil esté esperando autorización, la forma que usa
+`/spotify` para mostrarla es:
+
+```bash
+curl -s 127.0.0.1:3678/auth/code
+curl -s 127.0.0.1:3679/auth/code
+```
+
+Cada respuesta contiene la URL de Spotify y un código temporal. Jaime lo aprueba con la cuenta
+de Diego; Pilar, con la de Camila. La credencial no se pone en YAML: go-librespot la guarda en
+`state.json` bajo la carpeta privada del perfil.
+
+## Desvincular
+
+Detén sólo la instancia correspondiente, borra su credencial guardada y vuelve a iniciarla:
+
+```bash
+systemctl --user stop spotify@jaime && rm /opt/nodo/datos/spotify/jaime/state.json && systemctl --user start spotify@jaime
+```
+
+Para Pilar se sustituye `jaime` por `pilar`. Esto no borra `config.yml`; al reiniciar vuelve a
+entregar un código nuevo. No borres la carpeta: también contiene el lockfile mientras la unidad
+está activa.

+def leer_yaml_simple(ruta: Path) -> dict:
+    """Lee el subconjunto YAML de estas plantillas, sin dependencia externa."""
+    raiz: dict = {}
+    actual = raiz
+    for linea in ruta.read_text().splitlines():
+        if not linea or linea.lstrip().startswith("#"):
+            continue
+        sangria = len(linea) - len(linea.lstrip())
+        clave, separador, valor = linea.strip().partition(":")
+        if not separador:
+            raise ValueError(f"YAML inválido en {ruta.name}: {linea!r}")
+        if sangria == 0:
+            actual = raiz
+        elif sangria != 2:
+            raise ValueError(f"sangría no admitida en {ruta.name}: {linea!r}")
+        if valor.strip() == "":
+            dato = {}
+        elif valor.strip() in ("true", "false"):
+            dato = valor.strip() == "true"
+        elif valor.strip().isdigit():
+            dato = int(valor.strip())
+        elif valor.strip().startswith('"') and valor.strip().endswith('"'):
+            dato = valor.strip()[1:-1]
+        else:
+            dato = valor.strip()
+        actual[clave] = dato
+        if isinstance(dato, dict):
+            actual = dato
+    return raiz
+
+
+def validar(valor, esquema, ruta="config"):
+    tipo = esquema.get("type")
+    if tipo == "object":
+        if not isinstance(valor, dict):
+            raise AssertionError(f"{ruta}: se esperaba objeto")
+        props = esquema.get("properties", {})
+        if esquema.get("additionalProperties") is False:
+            extras = set(valor) - set(props)
+            if extras:
+                raise AssertionError(f"{ruta}: claves no admitidas: {sorted(extras)}")
+        for clave, dato in valor.items():
+            if clave not in props:
+                raise AssertionError(f"{ruta}: clave fuera del esquema: {clave}")
+            validar(dato, props[clave], f"{ruta}.{clave}")
+    elif tipo == "string" and not isinstance(valor, str):
+        raise AssertionError(f"{ruta}: se esperaba texto")
+    elif tipo == "boolean" and type(valor) is not bool:
+        raise AssertionError(f"{ruta}: se esperaba booleano")
+    elif tipo == "integer" and (type(valor) is not int):
+        raise AssertionError(f"{ruta}: se esperaba entero")
+    if "enum" in esquema and valor not in esquema["enum"]:
+        raise AssertionError(f"{ruta}: {valor!r} no pertenece a {esquema['enum']}")
+    if "minimum" in esquema and valor < esquema["minimum"]:
+        raise AssertionError(f"{ruta}: menor que {esquema['minimum']}")
+

+def abrir_directorio(ruta: Path, crear: bool = False) -> int:
+    """Abre cada tramo por descriptor, sin seguir enlaces que un usuario pueda intercambiar.
+
+    El descriptor resultante ancla las operaciones posteriores aunque alguien renombre una
+    carpeta mientras se prepara el despliegue. Sólo se llama después de validar la ruta textual.
+    """
+    if not ruta.is_absolute():
+        raise ValueError("directorio no absoluto")
+    fd = os.open("/", os.O_RDONLY | os.O_DIRECTORY)
+    try:
+        for tramo in ruta.parts[1:]:
+            try:
+                nuevo = os.open(tramo, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
+            except FileNotFoundError:
+                if not crear:
+                    raise
+                try:
+                    os.mkdir(tramo, 0o755, dir_fd=fd)
+                except FileExistsError:
+                    pass
+                nuevo = os.open(tramo, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
+            os.close(fd)
+            fd = nuevo
+        return fd
+    except Exception:
+        os.close(fd)
+        raise

+def leer_destino(destino: Path) -> bytes | None:
+    """Lee un destino existente por descriptores; un enlace nunca llega al diff ni al respaldo."""
+    try:
+        padre = abrir_directorio(destino.parent)
+    except FileNotFoundError:
+        return None
+    try:
+        try:
+            fd = os.open(destino.name, os.O_RDONLY | os.O_NOFOLLOW, dir_fd=padre)
+        except FileNotFoundError:
+            return None
+        try:
+            st = os.fstat(fd)
+            if not stat.S_ISREG(st.st_mode) or st.st_nlink != 1:
+                raise ValueError(f"destino no es archivo regular seguro: {destino}")
+            with os.fdopen(fd, "rb", closefd=False) as fh:
+                return fh.read()
+        finally:
+            os.close(fd)
+    finally:
+        os.close(padre)
